# Z04 주행(R) 시연: R1 → R2 → R3 → R4 → R5
# '#@' 줄은 지금 검증하는 요구사항이다. 시뮬레이터 이벤트 로그에 빨간색으로 표시된다.
# 시작 상태: 시동 OFF, 기어 P, 속도 0, 조도 30000, 강우 0%, 스위치 OFF, 주행 모델 OFF(속도 직접 입력)
# 시연 문서에 시간이 정해진 구간은 그대로 두고, 시간이 없는 구간은 단계 사이에 1~2초 대기를 넣었다.

# ===== R 주행 (가속 보통 +10, 풀 +25, 브레이크 -40, 타력 -5 km/h/s) =====
drive on

#@ R1 — 시동 OFF면 안 움직임, 시동은 정지 상태에서만 켜고 끔
ign on                      # 1. 시동 ON, 기어 D
wait 1
gear d
wait 1
ign off                     #    시동 OFF 후 가속
wait 1
pedal 1 0
wait 2
expect speed 0              #    가속해도 속도 0
pedal 0 0
wait 1
ign on                      # 2. 시동 ON, 속도 30까지 가속
wait 1
pedal 2 0
wait 1.2
pedal 0 0
expect speed 30
ign off                     #    주행 중 시동 OFF 시도
expect ign on               #    무시되고 ON 유지
wait 1
pedal 0 1                   # 3. 브레이크로 정지
wait 1.5
expect speed 0
ign off                     #    정지 후 시동 OFF
expect ign off              #    시동이 꺼짐
wait 2

#@ R2 — 기어 변경은 속도 0에서만
ign on
wait 1
pedal 2 0                   # 기어 D로 속도 30까지 가속
wait 1.2
pedal 0 0
expect speed 30
gear r
expect gear d               # 주행 중: D 그대로
wait 0.5
gear n
expect gear d
wait 0.5
gear p
expect gear d
wait 0.5
pedal 0 1                   # 브레이크로 정지
wait 1.5
expect speed 0
gear r
expect gear r               # 속도 0: R로 바뀜
wait 2                      # 다시 브레이크로 정지 상태

#@ R3 — D 전진, R 후진, P·N 정지
gear p
wait 1
pedal 1 0                   # P에서 가속 1초
wait 1
expect speed 0              # P: 속도 0
gear n                      # N에서 가속 1초
wait 1
expect speed 0              # N: 속도 0
gear d                      # D에서 가속 1초
wait 1
expect speed 10             # D: 전진 속도 증가
pedal 0 1                   # 브레이크로 정지
wait 1
expect speed 0
gear r                      # R에서 가속 1초
wait 1
pedal 1 0
wait 1
expect speed 10             # R: 후진 속도 증가
pedal 0 1                   # 브레이크로 정지
wait 1
expect speed 0
wait 2

#@ R4 — 브레이크 감속, 관성 감속, 브레이크 우선
gear d
wait 1
pedal 2 0                   # 기어 D, 속도 60까지 가속
wait 2.4
expect speed 60
pedal 0 0                   # 1. 가속·브레이크 모두 뗌 (2초)
wait 2
expect speed 50             #    천천히 줄어듦: 60 → 50 (-5 km/h/s)
pedal 1 1                   # 2. 가속과 브레이크 동시에 (0.5초)
wait 0.5
expect speed 30             #    가속을 밟고 있어도 줄어듦: 50 → 30 (-40 km/h/s)
pedal 0 1                   # 3. 브레이크만, 정지까지
wait 0.75
expect speed 0              #    1보다 빠르게 줄어 0에서 멈춤
wait 2
expect speed 0              #    정지 후에도 밟고 있어도 후진·음수 없음
expect gear d
wait 2

#@ R5 — 최고 속도와 가속 특성
pedal 2 0                   # 기어 D에서 풀 가속
wait 3.95
expect speed 99             # 0 → 100 km/h: 3.95초에 98.75
wait 0.05
expect speed 100            #               4.00초에 100 도달 (+25 km/h/s)
wait 6
expect speed 250            # 전진 최고 250
wait 1
expect speed 250            # 1초 유지: 더 오르지 않음
pedal 0 1                   # 브레이크
wait 1
expect speed 210            # 브레이크 감속률 -40 km/h/s
wait 5.25
expect speed 0
wait 1
gear r                      # 기어 R에서 풀 가속
wait 1
pedal 2 0
wait 0.8
expect speed 20             # 후진 최고 20
wait 1
expect speed 20             # 1초 유지: 더 오르지 않음
pedal 0 1                   # 다시 브레이크
wait 0.5
expect speed 0
wait 1
pedal 0 0
wait 1
gear p
wait 1
ign off
